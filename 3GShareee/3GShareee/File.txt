- (void)viewDidLoad {
    [super viewDidLoad];
    
    self.title = @"基本信息";
    self.view.backgroundColor = [UIColor colorWithWhite:0.95 alpha:1.0];
    
    [self setupUI];
    [self loadUserData];
}

- (void)setupUI {
    // 基本信息标题
    UILabel *titleLabel = [[UILabel alloc] initWithFrame:CGRectMake(20, 20, self.view.bounds.size.width - 40, 30)];
    titleLabel.text = @"基本信息";
    titleLabel.font = [UIFont boldSystemFontOfSize:18];
    titleLabel.textColor = [UIColor darkGrayColor];
    [self.view addSubview:titleLabel];
    
    // 头像区域
    UIView *avatarContainer = [[UIView alloc] initWithFrame:CGRectMake(0, 60, self.view.bounds.size.width, 100)];
    avatarContainer.backgroundColor = [UIColor whiteColor];
    [self.view addSubview:avatarContainer];
    
    // 头像标题
    UILabel *avatarTitle = [[UILabel alloc] initWithFrame:CGRectMake(20, 15, 100, 20)];
    avatarTitle.text = @"头像";
    avatarTitle.font = [UIFont systemFontOfSize:16];
    avatarTitle.textColor = [UIColor darkGrayColor];
    [avatarContainer addSubview:avatarTitle];
    
    // 头像图片
    _avatarImageView = [[UIImageView alloc] initWithFrame:CGRectMake(self.view.bounds.size.width - 90, 10, 70, 70)];
    _avatarImageView.contentMode = UIViewContentModeScaleAspectFill;
    _avatarImageView.clipsToBounds = YES;
    _avatarImageView.layer.cornerRadius = 35;
    _avatarImageView.layer.borderWidth = 1;
    _avatarImageView.layer.borderColor = [UIColor colorWithWhite:0.9 alpha:1.0].CGColor;
    [avatarContainer addSubview:_avatarImageView];
    
    // 昵称区域
    UIView *nicknameContainer = [self createInfoContainerWithY:CGRectGetMaxY(avatarContainer.frame) + 10 title:@"昵称"];
    _nicknameLabel = [self createValueLabelForContainer:nicknameContainer];
    [self.view addSubview:nicknameContainer];
    
    // 签名区域
    UIView *signatureContainer = [self createInfoContainerWithY:CGRectGetMaxY(nicknameContainer.frame) + 1 title:@"签名"];
    _signatureLabel = [self createValueLabelForContainer:signatureContainer];
    _signatureLabel.numberOfLines = 0;
    [self.view addSubview:signatureContainer];
    
    // 性别区域
    UIView *genderContainer = [self createInfoContainerWithY:CGRectGetMaxY(signatureContainer.frame) + 1 title:@"性别"];
    
    // 性别选择按钮
    _maleButton = [UIButton buttonWithType:UIButtonTypeCustom];
    _maleButton.frame = CGRectMake(self.view.bounds.size.width - 120, 15, 40, 20);
    [_maleButton setTitle:@"男" forState:UIControlStateNormal];
    [_maleButton setTitleColor:[UIColor colorWithRed:0.1 green:0.5 blue:0.9 alpha:1.0] forState:UIControlStateNormal];
    [_maleButton setImage:[UIImage systemImageNamed:@"checkmark.circle.fill"] forState:UIControlStateNormal];
    [_maleButton setTintColor:[UIColor colorWithRed:0.1 green:0.5 blue:0.9 alpha:1.0]];
    _maleButton.titleLabel.font = [UIFont systemFontOfSize:16];
    _maleButton.imageEdgeInsets = UIEdgeInsetsMake(0, -10, 0, 0);
    [genderContainer addSubview:_maleButton];
    
    _femaleButton = [UIButton buttonWithType:UIButtonTypeCustom];
    _femaleButton.frame = CGRectMake(self.view.bounds.size.width - 60, 15, 40, 20);
    [_femaleButton setTitle:@"女" forState:UIControlStateNormal];
    [_femaleButton setTitleColor:[UIColor grayColor] forState:UIControlStateNormal];
    [_femaleButton setImage:[UIImage systemImageNamed:@"circle"] forState:UIControlStateNormal];
    [_femaleButton setTintColor:[UIColor grayColor]];
    _femaleButton.titleLabel.font = [UIFont systemFontOfSize:16];
    _femaleButton.imageEdgeInsets = UIEdgeInsetsMake(0, -10, 0, 0);
    [genderContainer addSubview:_femaleButton];
    
    [self.view addSubview:genderContainer];
    
    // 邮箱区域
    UIView *emailContainer = [self createInfoContainerWithY:CGRectGetMaxY(genderContainer.frame) + 1 title:@"邮箱"];
    _emailLabel = [self createValueLabelForContainer:emailContainer];
    [self.view addSubview:emailContainer];
}

- (UIView *)createInfoContainerWithY:(CGFloat)y title:(NSString *)title {
    UIView *container = [[UIView alloc] initWithFrame:CGRectMake(0, y, self.view.bounds.size.width, 50)];
    container.backgroundColor = [UIColor whiteColor];
    
    UILabel *titleLabel = [[UILabel alloc] initWithFrame:CGRectMake(20, 15, 100, 20)];
    titleLabel.text = title;
    titleLabel.font = [UIFont systemFontOfSize:16];
    titleLabel.textColor = [UIColor darkGrayColor];
    [container addSubview:titleLabel];
    
    return container;
}

- (UILabel *)createValueLabelForContainer:(UIView *)container {
    UILabel *label = [[UILabel alloc] initWithFrame:CGRectMake(self.view.bounds.size.width - 220, 15, 180, 20)];
    label.textAlignment = NSTextAlignmentRight;
    label.font = [UIFont systemFontOfSize:16];
    label.textColor = [UIColor darkGrayColor];
    [container addSubview:label];
    return label;
}

- (void)loadUserData {
    // 加载用户数据 - 这里使用示例数据
    _avatarImageView.image = [UIImage imageNamed:@"user_avatar"]; // 替换为实际头像图片
    _nicknameLabel.text = @"share小白";
    _signatureLabel.text = @"开心了就笑，不开心了就待会儿再笑";
    _emailLabel.text = @"186****3@qq.com";
    
    // 设置性别选择状态
    [self updateGenderSelection:YES]; // 默认选中男
}

- (void)updateGenderSelection:(BOOL)isMale {
    if (isMale) {
        [_maleButton setImage:[UIImage systemImageNamed:@"checkmark.circle.fill"] forState:UIControlStateNormal];
        [_maleButton setTintColor:[UIColor colorWithRed:0.1 green:0.5 blue:0.9 alpha:1.0]];
        [_maleButton setTitleColor:[UIColor colorWithRed:0.1 green:0.5 blue:0.9 alpha:1.0] forState:UIControlStateNormal];
        
        [_femaleButton setImage:[UIImage systemImageNamed:@"circle"] forState:UIControlStateNormal];
        [_femaleButton setTintColor:[UIColor grayColor]];
        [_femaleButton setTitleColor:[UIColor grayColor] forState:UIControlStateNormal];
    } else {
        [_femaleButton setImage:[UIImage systemImageNamed:@"checkmark.circle.fill"] forState:UIControlStateNormal];
        [_femaleButton setTintColor:[UIColor colorWithRed:0.1 green:0.5 blue:0.9 alpha:1.0]];
        [_femaleButton setTitleColor:[UIColor colorWithRed:0.1 green:0.5 blue:0.9 alpha:1.0] forState:UIControlStateNormal];
        
        [_maleButton setImage:[UIImage systemImageNamed:@"circle"] forState:UIControlStateNormal];
        [_maleButton setTintColor:[UIColor grayColor]];
        [_maleButton setTitleColor:[UIColor grayColor] forState:UIControlStateNormal];
    }
}
